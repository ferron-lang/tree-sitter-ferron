use pkg.a.{Status as AStatus, Ok as AOk, make as make_a}
use pkg.b.B as _

pub record Box[T] {
  value: T
}

trait Unwrap {
  type Item
  fn unwrap(self) -> Self.Item
}

impl[T] Unwrap for Box[T] {
  type Item = T

  fn unwrap(self) -> T {
    return self.value
  }
}

pub fn apply[T](value: T, f: fn(T) -> u16) -> u16
  where T: Iterable[Item = u16]
{
  let updated = Box[u16] {
    ..base,
    value: 1u16,
  }

  match f(value) {
    Some(result) => return result
    None => return updated.value
  }
}
