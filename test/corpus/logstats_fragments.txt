fn analyze_file!(
  fs: FsRead,
  path: Path,
) -> Result[Stats, AppError]
  uses [fs.read]
{
  with file: ReadFile =
    fs.open_read!(path)
      .map_error(|cause| ReadFailed(path, cause))?
  {
    let text =
      file.read_to_end!()
        .map_error(|cause| ReadFailed(path, cause))?

    Ok(analyze_text(text))
  }
}

fn describe_error(error: AppError) -> String {
  match error {
    Usage(message) =>
      message

    ReadFailed(path, cause) =>
      "cannot read"
  }
}
